// ============================================================
// 거래(홍보) 게시판용 Firestore 규칙
//
// 적용 방법 (Firebase 콘솔)
//  1) Authentication > 로그인 방법 > "익명" 사용 설정
//  2) Firestore Database > 규칙 에서, 기존 규칙(guides, pageStats 등)은 그대로 두고
//     아래 함수 2개와 match 블록들을 `match /databases/{database}/documents { ... }` 안에 "추가"한 뒤 게시
//  3) 관리자 지정: 사이트에서 거래 게시판을 한 번 열면 Authentication > 사용자 목록에 익명 계정이 생긴다.
//     관리자로 쓸 계정의 UID를 복사해, Firestore에 컬렉션 admins, 문서 ID = 그 UID 로 문서를 하나 만든다(필드는 아무거나).
//
// 이 규칙이 막는 것
//  - 로그인(익명 포함) 없이 쓰기, 남의 글/댓글 수정·삭제 (본인과 관리자만 가능)
//  - 링크(http, www, .com 등)와 현거래 단어(현금, 계좌, 입금, 상품권 등)가 들어간 글과 댓글
//  - 정해진 필드 이외의 값, 너무 긴 글, 다른 사람 UID로 작성
// 사이트 코드(src/trade/validate.ts)에도 같은 검사가 있지만, 코드는 우회할 수 있으므로 서버 규칙이 실제 방어선이다.
// ============================================================

    function isAdmin() {
      return request.auth != null
        && exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }

    function cleanText(s) {
      return !s.matches('(?is).*(https?:|ftp:|www\\.|\\.(com|net|kr|co|io|me|gg|tv|xyz|link|org|be|ly|app|dev|info|site|shop|cc|to|ai)).*')
        && !s.matches('(?is).*(현거래|현금|계좌|입금|송금|이체|원화|페이팔|paypal|카카오페이|네이버페이|상품권|기프티콘|비트코인|리얼머니).*');
    }

    function okStr(v, minLen, maxLen) {
      return v is string && v.size() >= minLen && v.size() <= maxLen && cleanText(v);
    }

    match /admins/{uid} {
      allow read: if request.auth != null && request.auth.uid == uid;
      allow write: if false;
    }

    match /trades/{tradeId} {
      allow read: if true;

      allow create: if request.auth != null
        && request.resource.data.keys().hasOnly(['category', 'title', 'body', 'price', 'gameNickname', 'location', 'saleTime', 'status', 'ownerUid', 'createdAt'])
        && request.resource.data.keys().hasAll(['category', 'title', 'body', 'price', 'gameNickname', 'location', 'saleTime', 'status', 'ownerUid', 'createdAt'])
        && request.resource.data.ownerUid == request.auth.uid
        && request.resource.data.category in ['equip', 'material', 'socket', 'wanted']
        && request.resource.data.status == 'selling'
        && request.resource.data.createdAt == request.time
        && okStr(request.resource.data.title, 2, 40)
        && okStr(request.resource.data.body, 5, 500)
        && okStr(request.resource.data.price, 1, 30)
        && okStr(request.resource.data.gameNickname, 1, 20)
        && okStr(request.resource.data.location, 1, 40)
        && okStr(request.resource.data.saleTime, 1, 40);

      allow update: if request.auth != null
        && (resource.data.ownerUid == request.auth.uid || isAdmin())
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'updatedAt'])
        && request.resource.data.status in ['selling', 'reserved', 'done'];

      allow delete: if request.auth != null
        && (resource.data.ownerUid == request.auth.uid || isAdmin());

      match /comments/{commentId} {
        allow read: if true;

        allow create: if request.auth != null
          && request.resource.data.keys().hasOnly(['nickname', 'text', 'ownerUid', 'createdAt'])
          && request.resource.data.keys().hasAll(['nickname', 'text', 'ownerUid', 'createdAt'])
          && request.resource.data.ownerUid == request.auth.uid
          && request.resource.data.createdAt == request.time
          && okStr(request.resource.data.nickname, 1, 20)
          && okStr(request.resource.data.text, 1, 200)
          && exists(/databases/$(database)/documents/trades/$(tradeId));

        allow delete: if request.auth != null
          && (resource.data.ownerUid == request.auth.uid
              || isAdmin()
              || get(/databases/$(database)/documents/trades/$(tradeId)).data.ownerUid == request.auth.uid);

        allow update: if false;
      }
    }
